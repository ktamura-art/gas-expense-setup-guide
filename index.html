/** 1ファイルの処理 */
function processFile_(file) {
  const name = file.getName();

  // ---- ① 判定（発行元・日付・金額） ----
  const text = extractText_(file);
  const vendor = detectVendor_(name + '\n' + text);
  const date   = detectDate_(name + '\n' + text);
  const amount = detectAmount_(text) || detectAmountFromName_(name);

  if (!date || !amount) {
    return {
      status: 'エラー',
      row: [now_(), name, vendor || '', date ? formatDate_(date) : '', amount || '',
            '', '', '', '', '', '', 'エラー', '日付または金額を判定できません'],
    };
  }

  // ---- ② カレンダー突合 ----
  const matched = matchCalendar_(date, vendor);
  // 区間は「領収書から抽出」を優先し、取れない場合はカレンダーの場所で補完
  const rawRoute = detectRoute_(text) || matched.route || '';
  // ★表記ゆれ辞書があれば、区間を正式表記に揃えてから料金を引く
  const route = normalizeRoute_(rawRoute);

  // ---- ③ 区間料金の引き当て（案②：区間マスタ） ----
  const fare = route ? lookupFare_(route) : null;

  // ---- ①' リネーム ----
  const category = guessCategory_(vendor, matched);
  const newName = [formatDate_(date), vendor || 'NA', amount, category].join('_') + getExt_(name);
  file.setName(newName);

  // ---- ④ MF経費へ連携（★運賃・区間もあわせて転記：2026-08-10の決定） ----
  let mfId = '';
  if (CONFIG.ENABLE_MF) {
    mfId = createMfExpense_({ date: date, amount: amount, vendor: vendor,
                              memo: matched.memo, category: category,
                              route: route, fare: fare, file: file });
  }

  const status = matched.candidates > 1 || !matched.memo ? '要確認' : '正常';
  const note = matched.candidates > 1 ? ('候補が複数：' + matched.candidateTitles) :
               (!matched.memo ? '該当予定なし（摘要は手入力）' : (fare ? ('区間マスタ参照:' + fare) : ''));

  return {
    status: status,
    row: [
      now_(), newName, vendor, formatDate_(date), amount, category,
      route, fare || '', matched.memo || '', matched.candidateTitles || '', mfId, status, note,
    ],
  };
}


/**
 * 区間の表記を正式表記に揃える（例：東京駅→大分空港 ＝ 東京→大分）
 * 「経費自動化_表記ゆれ辞書.gs」を入れている場合のみ辞書を使い、
 * 入れていない場合はそのまま返します。
 */
function normalizeRoute_(route) {
  if (!route) return '';
  if (typeof normalizeTerm_ !== 'function') return route;

  const m = String(route).match(/^(.+?)\s*(?:→|->|〜|~|ー|-)\s*(.+)$/);
  if (!m) return normalizeTerm_('', '地名', route);
  return normalizeTerm_('', '地名', m[1]) + '→' + normalizeTerm_('', '地名', m[2]);
}

/**
 * PDFからテキスト抽出（GoogleドキュメントへOCR変換して読む）
 * ★共有ドライブ（マイドライブ以外）にも対応するため supportsAllDrives を指定
 */
function extractText_(file) {
  try {
    if (file.getMimeType() !== MimeType.PDF) return '';
    const doc = Drive.Files.copy(
      { title: 'tmp_ocr_' + file.getName(), mimeType: MimeType.GOOGLE_DOCS },
      file.getId(),
      { ocr: true, ocrLanguage: 'ja', supportsAllDrives: true, supportsTeamDrives: true });
    const text = DocumentApp.openById(doc.id).getBody().getText();
    DriveApp.getFileById(doc.id).setTrashed(true);   // 一時ファイルを削除
    return text;
  } catch (e) {
    Logger.log('OCRスキップ: ' + e);
    return '';
  }
}

/**
 * ★文字の正規化（最重要）
 * 一部の領収書（JR九州など）は「⽉ ⽇ ⾦ ⼈」といった互換文字を使っており、
 * 通常の「月 日 金 人」と別コードのため、そのままでは正規表現が一致しない。
 * NFKC正規化で通常文字に揃えてから判定する。
 */
function norm_(t) { return String(t).normalize('NFKC'); }

function detectVendor_(text) {
  text = norm_(text);
  for (const v of VENDORS) {
    if (v.keywords.some(function (k) { return text.indexOf(k) >= 0; })) return v.name;
  }
  return '';
}

/**
 * 日付の検出：★「発行日」ではなく「利用日（乗車日・搭乗日・チェックイン日）」を優先
 * カレンダー突合の精度は、この利用日が正しく取れるかで決まる。
 */
function detectDate_(text) {
  text = norm_(text);
  const mk = function (y, mo, d) { return new Date(Number(y), Number(mo) - 1, Number(d)); };
  let m;

  // ① 明示ラベル付きの利用日
  m = text.match(/(?:乗車日|搭乗日|利用日|ご利用日|チェックイン日?)[^0-9]{0,20}(20\d{2})\s?[年\/\-\.]\s?(\d{1,2})\s?[月\/\-\.]\s?(\d{1,2})/);
  if (m) return mk(m[1], m[2], m[3]);

  // ② 英語表記のチェックイン日（例：Arrival: December 13, 2025）
  m = text.match(/(?:Arrival|チェックイン)[^A-Za-z0-9]{0,30}(January|February|March|April|May|June|July|August|September|October|November|December)\s+(\d{1,2}),\s*(20\d{2})/);
  if (m) return mk(m[3], MON_FULL[m[1]], m[2]);

  // ③ 航空券の旅程日付（例：03OCT2026）※発行日を拾わないようDEPARTURE以降に限定
  const dep = text.indexOf('DEPARTURE');
  const scope = dep >= 0 ? text.slice(dep) : text;
  m = scope.match(/(\d{2})(JAN|FEB|MAR|APR|MAY|JUN|JUL|AUG|SEP|OCT|NOV|DEC)(20\d{2})/);
  if (m) return mk(m[3], MON_ABBR[m[2]], m[1]);

  // ④ 「YYYY年M月D日（曜）」＝旅程表・明細行に多い形式
  m = text.match(/(20\d{2})[年]\s?(\d{1,2})[月]\s?(\d{1,2})[日]\s*[（(][日月火水木金土]/);
  if (m) return mk(m[1], m[2], m[3]);

  // ⑤ フォールバック（発行日など）
  m = text.match(/(20\d{2})[年]\s?(\d{1,2})[月]\s?(\d{1,2})/);
  if (m) return mk(m[1], m[2], m[3]);
  m = text.match(/(20\d{2})\/(\d{1,2})\/(\d{1,2})/);
  if (m) return mk(m[1], m[2], m[3]);
  return null;
}

/**
 * 金額の検出：合計・TOTALの近傍を優先（JPY表記にも対応）
 * ★近傍に「JPY0（手数料）」等が混在するため、近傍の最大値を採用する
 */
function detectAmount_(text) {
  text = norm_(text);
  let m = text.match(/(?:合計金額|ご利用金額|合計|TOTAL)([\s\S]{0,120})/);
  if (m) {
    const cands = (m[1].match(/(?:JPY|¥|￥)\s?([0-9][0-9,]{2,})/g) || [])
      .map(function (x) { return Number(x.replace(/[^0-9]/g, '')); })
      .filter(function (n) { return n > 0; });
    if (cands.length) return Math.max.apply(null, cands);
    const plain = m[1].match(/([0-9][0-9,]{2,})\s*円/);
    if (plain) return Number(plain[1].replace(/,/g, ''));
  }
  m = text.match(/(?:金額|請求金額|領収金額|お支払[い]?金額)[^0-9]{0,10}(?:JPY|¥|￥)?\s?([0-9][0-9,]{2,})/);
  if (m) return Number(m[1].replace(/,/g, ''));
  const all = (text.match(/(?:JPY|¥|￥)\s?([0-9][0-9,]{2,})/g) || [])
    .map(function (s) { return Number(s.replace(/[^0-9]/g, '')); });

function matchCalendar_(date, vendor) {

    ? CalendarApp.getDefaultCalendar()
    : CalendarApp.getCalendarById(CONFIG.CALENDAR_ID);
  if (!cal) return { memo: '', candidates: 0, route: '', candidateTitles: '' };

  const start = new Date(date); start.setHours(0, 0, 0, 0);

  if (events.length === 0) return { memo: '', candidates: 0, route: '', candidateTitles: '' };

  // 区分に応じて優先度を付ける
  let picked;
  if (vendor === 'ホテル' || vendor === 'JAL' || vendor === 'ANA' || vendor === 'JR' || vendor === 'スカイマーク') {
    picked = events.filter(function (e) { return e.getLocation(); })[0] || events[0];   // 場所ありを優先
  } else {
    picked = events.filter(function (e) { return e.getStartTime().getHours() >= 18 && e.getGuestList().length >= 2; })[0]
             || events[0];   // 会食＝夜間＋ゲスト2名以上
  }

  const route = picked.getLocation() || '';
  const memo = [picked.getTitle(), route ? ('（' + route + '）') : ''].join('') + 'のため';

    memo: memo,
    route: route,
    candidates: events.length,
    candidateTitles: events.map(function (e) { return e.getTitle(); }).join(' / '),
  };
}

/**
 * 実行日から、処理すべき締め区間を返す
 *
 *   21日以降に実行  → 当月20日で終わる区間（＝ちょうど締まったばかりの区間）

  return {
    from: from,

function readProcessedReceipts_(from, to) {
  const sheet = getLogSheet_();
  const values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];

  const col = headerIndex_(values[0]);
  if (col['利用日'] === undefined) {
    throw new Error('処理ログに「利用日」の列が見つかりません。メイン側のLOG_HEADERSをご確認ください');
  }

  const fromStr = formatDateIso_(from);
  const toStr   = formatDateIso_(to);

  const pick = function (row, name) {
    return col[name] === undefined ? '' : row[col[name]];
  };

  const receipts = [];
  for (let i = 1; i < values.length; i++) {
    const row = values[i];
    const used = normalizeLogDate_(pick(row, '利用日'));
    if (!used) continue;
    if (used < fromStr || used > toStr) continue;

    receipts.push({
      dateStr: used,
      file:    String(pick(row, 'ファイル名') || ''),
      vendor:  String(pick(row, '発行元') || ''),
      amount:  pick(row, '金額'),
      route:   String(pick(row, '区間') || ''),
      fare:    pick(row, '運賃'),
      status:  String(pick(row, '状態') || ''),
    });
  }
  return receipts;
}

/** ログの日付セルを yyyy-MM-dd の文字列に揃える（日付型・文字列型の両方に対応） */
function normalizeLogDate_(v) {
  if (!v) return '';
  if (Object.prototype.toString.call(v) === '[object Date]') return formatDateIso_(v);
  const s = String(v).normalize('NFKC').trim();
  const m = s.match(/(\d{4})\D{0,2}(\d{1,2})\D{0,2}(\d{1,2})/);
  if (!m) return '';
  return m[1] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[3]).slice(-2);
}

  if (!m) {
    return { prefix: '', label: '', detail: t, from: '', to: '',
             isMove: false, isDining: false, category: null, matched: false };
  }
  const prefix = m[1].toLowerCase();
  const detail = (m[2] || '').trim();
  const def = CAL_PREFIX[prefix] || { label: '', category: null, isMove: false, isDining: false };

 * 駅すぱあとの Price は kind ごとに分かれるため、合計（Summary）を優先して採用する
 *   Summary       … 運賃＋料金の合計
 *   FareSummary   … 運賃のみ
 *   ChargeSummary … 特急料金など
 */
function pickTotalFare_(json) {
  try {
    let courses = json.ResultSet && json.ResultSet.Course;
    if (!courses) return null;
    if (!Array.isArray(courses)) courses = [courses];
    const course = courses[0];

    let prices = course.Price;
    if (!prices) return null;
    if (!Array.isArray(prices)) prices = [prices];

    const byKind = {};
    prices.forEach(function (p) {
      const oneway = Number(p.Oneway || p.oneway || 0);
      if (p.kind && oneway > 0) byKind[p.kind] = (byKind[p.kind] || 0) + oneway;
    });

    if (byKind['Summary']) return byKind['Summary'];
    const fare = byKind['FareSummary'] || 0;
    const charge = byKind['ChargeSummary'] || 0;
    if (fare + charge > 0) return fare + charge;
    return null;
  } catch (e) {
    Logger.log('運賃の抽出に失敗: ' + e);
    return null;
  }
}

/** ルートA：メールから領収書を収集して「領収書保管フォルダー」へ保存 */
function collectReceiptsFromMail() {
  const folder = DriveApp.getFolderById(MAIL_CONFIG.FOLDER_IN);
  const label = getOrCreateLabel_(MAIL_CONFIG.PROCESSED_LABEL);

  let saved = 0, noAttach = 0;

  MAIL_CONFIG.SOURCES.forEach(function (src) {
    const q = src.query +
      ' newer_than:' + MAIL_CONFIG.LOOKBACK_DAYS + 'd' +
      ' -label:' + MAIL_CONFIG.PROCESSED_LABEL;
    const threads = GmailApp.search(q, 0, 100);

    threads.forEach(function (thread) {
      thread.getMessages().forEach(function (msg) {
        const r = saveAttachments_(msg, src.name, folder);
        saved += r.saved;
        noAttach += r.noAttach;
      });
      thread.addLabel(label);   // 二重取込の防止
    });
  });

  Logger.log('メール収集：PDF保存 %s件 ／ 添付なし %s件', saved, noAttach);
  return { saved: saved, noAttach: noAttach };
}


/** 1通のメールからPDF添付を保存 */
function saveAttachments_(msg, vendorName, folder) {
  const pdfs = msg.getAttachments().filter(function (a) {
    return a.getContentType() === 'application/pdf';
  });

  if (pdfs.length === 0) {
    // 添付なし＝本文にDLリンクがある型。ログだけ残す（人が対応）
    Logger.log('添付なし: [%s] %s', vendorName, msg.getSubject());
    return { saved: 0, noAttach: 1 };
  }

  let saved = 0;
  pdfs.forEach(function (pdf) {
    // 仮の命名（正式なリネームはメイン側の判定処理で行う）
    const dateStr = Utilities.formatDate(msg.getDate(), 'Asia/Tokyo', 'yyyyMMdd');
    const name = [vendorName, dateStr, pdf.getName()].join('_');

    // 同名ファイルが既にあればスキップ（重複防止）
    if (folder.getFilesByName(name).hasNext()) return;

    folder.createFile(pdf.copyBlob()).setName(name);
    saved++;
  });

  return { saved: saved, noAttach: 0 };
}
